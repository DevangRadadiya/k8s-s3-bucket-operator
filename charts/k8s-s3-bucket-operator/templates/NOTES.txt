{{- if and .Values.minio.createSecret (not .Values.minio.existingSecret) }}

   kubectl create secret generic {{ .Values.minio.secretName }} -n {{ include "k8s-s3-bucket-operator.namespace" . }} \
     --from-literal=MINIO_ENDPOINT=... --from-literal=MINIO_ACCESS_KEY=... \
     --from-literal=MINIO_SECRET_KEY=... --from-literal=MINIO_USE_SSL=false \
     --dry-run=client -o yaml | kubectl apply -f -
   kubectl rollout restart deploy/{{ include "k8s-s3-bucket-operator.fullname" . }} -n {{ include "k8s-s3-bucket-operator.namespace" . }}
